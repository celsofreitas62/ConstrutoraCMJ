Private Sub Report_Open(Cancel As Integer)
    ' =====================
    ' Abrir Arquivo Sistema
    ' =====================
    
    Dim con As Object
    Dim rs As Object
    Dim stSql As String
    
    Set con = Application.CurrentProject.Connection
    stSql = "SELECT * FROM [Arquivo Sistema]"
    stSql = stSql & " WHERE [IDSistema] = 1"
    stSql = stSql & " ORDER BY [IDSistema];"
    Set rs = CreateObject("ADODB.Recordset")
    rs.Open stSql, con, 1
    
    If (rs.EOF) Then
        Me![NomeCliente].Caption = "Personalizar Cliente"
    Else
        Me![NomeCliente].Caption = rs![Cliente]
    End If

    rs.Close
    Set rs = Nothing
    Set con = Nothing
End Sub
