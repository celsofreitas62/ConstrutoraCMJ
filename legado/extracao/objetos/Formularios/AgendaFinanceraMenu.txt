Private Sub AbrirAgrupada_Click()
    Dim stDocName As String
    Dim stLinkCriteria As String
    Dim MovAgrupador
    
    If MovPagtoLote <> 0 Then
        MovAgrupador = Me![MovPagtoLote]
        stDocName = "AgrupadaLoteViaMovto"
        stLinkCriteria = "[MovPagtoLote]=" & MovAgrupador
    Else
        If AgrupadorVenda <> 0 Then
            MovAgrupador = Me![AgrupadorVenda]
            stDocName = "AgrupadaVendasViaMovto"
            stLinkCriteria = "[AgrupadorVenda]=" & MovAgrupador
        End If
    End If
    
    If MovAgrupador <> 0 Then DoCmd.OpenForm stDocName, , , stLinkCriteria
    
End Sub

Private Sub AbrirMovimento_Click()
On Error GoTo Err_AbrirMovimento_Click

    Dim stDocName As String
    Dim stLinkCriteria As String

    stDocName = "F Movimento Contas"
    
    stLinkCriteria = "[NumMovimento]=" & Me![NumMovimento]
    DoCmd.OpenForm stDocName, , , stLinkCriteria

Exit_AbrirMovimento_Click:
    Exit Sub

Err_AbrirMovimento_Click:
    MsgBox Err.Description
    Resume Exit_AbrirMovimento_Click

End Sub

Private Sub Confirmado_Click()
    Dim EstaConfirmado As Boolean
   
    EstaConfirmado = Confirmado
    If (EstaConfirmado) Then
        Efetuado = True
        If (IsNull(DataPagto)) Then DataPagto = Date
        ' se lançto é de venda atribui os códigos do parceiro ou empresa
        If (CodigoParceiro = 1) Then ' N/A
            If (CodAtribuiParceiro <> 0) Then
                CodAnteriorParceiro = CodigoParceiro
                CodigoParceiro = CodAtribuiParceiro
            End If
        End If
        If (EmpVinculada = 1) Then
            If (CodAtribuiEmpresa <> 0) Then
                CodAnteriorEmpresa = EmpVinculada
                EmpVinculada = CodAtribuiEmpresa
            End If
        End If
    Else
        DataPagto = Null
        If (CodigoParceiro <> 1) Then
            If (CodAnteriorParceiro <> 0) Then
                CodigoParceiro = CodAnteriorParceiro
            End If
        End If
        If (EmpVinculada <> 1) Then
            If (CodAnteriorEmpresa <> 0) Then
                EmpVinculada = CodAnteriorEmpresa
            End If
        End If
    End If

    ' grava registro
'    DoCmd.DoMenuItem acFormBar, acRecordsMenu, acSaveRecord, , acMenuVer70


'    DoCmd.GoToRecord , , acNext
    
'    DoCmd.GoToRecord , , acPrevious
    

    ' se tem pagto lote atualiza todos os movtos envolvidos para Confirmado e Efetuado = S ou N
    If (MovPagtoLote <> 0) Then
        DoCmd.OpenQuery "AtualizaEfetuadoLoteAgenda", acViewNormal, acEdit
        DoCmd.OpenQuery "AtualizaConfirmadoLoteAgenda", acViewNormal, acEdit
    End If
    If (LinkVenda <> 0) Then
        DoCmd.OpenQuery "AtualizaEfetuadoVendaAgenda", acViewNormal, acEdit
        DoCmd.OpenQuery "AtualizaConfirmadoVendaAgenda", acViewNormal, acEdit
    End If
    
End Sub
