Private Sub Form_Activate()
    Dim iMeses As Integer
    
    Me![AnoOcorrencia].Value = Me![UltimoAnoGerado]
    Me![ProximoMes].Value = Me![UltimoMesGerado]
    If Me!UltimoMesGerado = 1 Then Me!UltimoMesExtenso = "Janeiro"
    If Me!UltimoMesGerado = 2 Then Me!UltimoMesExtenso = "Fevereiro"
    If Me!UltimoMesGerado = 3 Then Me!UltimoMesExtenso = "Março"
    If Me!UltimoMesGerado = 4 Then Me!UltimoMesExtenso = "Abril"
    If Me!UltimoMesGerado = 5 Then Me!UltimoMesExtenso = "Maio"
    If Me!UltimoMesGerado = 6 Then Me!UltimoMesExtenso = "Junho"
    If Me!UltimoMesGerado = 7 Then Me!UltimoMesExtenso = "Julho"
    If Me!UltimoMesGerado = 8 Then Me!UltimoMesExtenso = "Agosto"
    If Me!UltimoMesGerado = 9 Then Me!UltimoMesExtenso = "Setembro"
    If Me!UltimoMesGerado = 10 Then Me!UltimoMesExtenso = "Outubro"
    If Me!UltimoMesGerado = 11 Then Me!UltimoMesExtenso = "Novembro"
    If Me!UltimoMesGerado = 12 Then
        Me!UltimoMesExtenso = "Dezembro"
        Me![ProximoMes].Value = 1
        Me![MesOcorrencia].Value = 1
        Me![AnoOcorrencia].Value = Me!UltimoAnoGerado + 1
    End If
    
    If Me!UltimoMesGerado <> 12 Then
        Me![MesOcorrencia].Value = Me!UltimoMesGerado + 1
        Me![ProximoMes].Value = Me![ProximoMes] + 1
    End If
    
    iMeses = 1
    Do While iMeses <= 12
        If (iMeses = 1 And iMeses < Me![MesOcorrencia]) Then Me![Mes1].Enabled = False
        If (iMeses = 2 And iMeses < Me![MesOcorrencia]) Then Me![Mes2].Enabled = False
        If (iMeses = 3 And iMeses < Me![MesOcorrencia]) Then Me![Mes3].Enabled = False
        If (iMeses = 4 And iMeses < Me![MesOcorrencia]) Then Me![Mes4].Enabled = False
        If (iMeses = 5 And iMeses < Me![MesOcorrencia]) Then Me![Mes5].Enabled = False
        If (iMeses = 6 And iMeses < Me![MesOcorrencia]) Then Me![Mes6].Enabled = False
        If (iMeses = 7 And iMeses < Me![MesOcorrencia]) Then Me![Mes7].Enabled = False
        If (iMeses = 8 And iMeses < Me![MesOcorrencia]) Then Me![Mes8].Enabled = False
        If (iMeses = 9 And iMeses < Me![MesOcorrencia]) Then Me![Mes9].Enabled = False
        If (iMeses = 10 And iMeses < Me![MesOcorrencia]) Then Me![Mes10].Enabled = False
        If (iMeses = 11 And iMeses < Me![MesOcorrencia]) Then Me![Mes11].Enabled = False
        If (iMeses = 12 And iMeses < Me![MesOcorrencia]) Then Me![Mes12].Enabled = False
        iMeses = iMeses + 1
    Loop
End Sub

Private Sub GerarOcorrencia_Click()
On Error GoTo GerarOcorrencia_Click_Err

    Dim Mensagem, Estilo, Resposta
    Mensagem = "Confirma Gerar Ocorrencias no Mes ?"
    Estilo = vbYesNo + vbQuestion + vbDefaultButton2
    Resposta = MsgBox(Mensagem, Estilo)
    
    If Resposta = vbNo Then
       GoTo GerarOcorrencia_Click_Exit
    End If
    
    Me![ProximoMes] = Me![MesOcorrencia]
    DoCmd.OpenQuery "Gerar AnoMes Ocorrencia", acViewNormal, acAdd
    DoCmd.OpenQuery "Gerar Ocorrencias Mensais no Movto", acViewNormal, acAdd
    
    MsgBox "Movimento gerado com sucesso !"

GerarOcorrencia_Click_Exit:
    DoCmd.Close
    Exit Sub

GerarOcorrencia_Click_Err:
    MsgBox Error$
    Resume GerarOcorrencia_Click_Exit

End Sub

Private Sub Form_Load()
    
    Dim iMeses
    Dim strPrimeiroMes As String
    Dim PrimeiroAno As String
    Dim Conexao As Object
    Dim Arquivo As Object
    Dim Selecao As String
    Dim MesValido
    Dim Mensagem, Estilo, Resposta
    
    Set Conexao = Application.CurrentProject.Connection
    Selecao = "SELECT * FROM [AnoMes Ocorrencias Mensais]"
    Set Arquivo = CreateObject("ADODB.Recordset")
    Arquivo.Open Selecao, Conexao, 1
        
    If (Arquivo.EOF) Then
        Me![AnoOcorrencia].Value = Year(Date)
        Me![ProximoMes].Value = (Month(Date) - 1) 'PrimeiroMes
        
        DoCmd.OpenQuery "Gerar AnoMes Ocorrencia", acViewNormal, acAdd
        
        Arquivo.Close
        Set Arquivo = Nothing
        Set Conexao = Nothing
        DoCmd.Close
        DoCmd.OpenForm "F Gerar Ocorrencias Mensais"
    Else
        Arquivo.Close
        Set Arquivo = Nothing
        Set Conexao = Nothing
        DoCmd.GoToRecord , , acLast
    End If
    
End Sub
