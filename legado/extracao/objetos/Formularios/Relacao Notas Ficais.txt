Private Sub AbrirExtratoEmpresa_Click()
On Error GoTo Err_AbrirExtratoEmpresa_Click

    Dim stDocName As String

    stDocName = "Notas Fiscais Periodo"
    DoCmd.OpenReport stDocName, acPreview

Exit_AbrirExtratoEmpresa_Click:
    Exit Sub

Err_AbrirExtratoEmpresa_Click:
    MsgBox Err.Description
    Resume Exit_AbrirExtratoEmpresa_Click
    
End Sub
Private Sub Form_Activate()
    DTFinal = Date
    Do While Day(DTFinal) <= 31
        DTFinal = DTFinal + 1
       If Month(DTFinal) <> Month(Date) Then
            DTFinal = DTFinal - 1
            Exit Do
        End If
    Loop
    DTInicial = DateSerial(Year(Date), Month(Date), ((Day(Date)) - (Day(Date)) + 1))
End Sub
