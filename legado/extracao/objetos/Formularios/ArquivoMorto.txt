Private Sub ExecutarRelatorio_Click()
On Error GoTo Err_ExecutarRelatorio_Click

    DoCmd.OpenQuery "ArquivoMorto Arquivar", acViewNormal, acAdd
    DoCmd.OpenQuery "ArquivoMorto Gerar Mov Acum", acViewNormal, acAdd
    DoCmd.OpenQuery "ArquivoMorto Excluir Mov", acViewNormal, acEdit
    
    MsgBox "Movimento arquivado com sucesso !", vbInformation, "CashBook"

Exit_ExecutarRelatorio_Click:
    Exit Sub

Err_ExecutarRelatorio_Click:
    MsgBox Err.Description
    Resume Exit_ExecutarRelatorio_Click
    
End Sub
