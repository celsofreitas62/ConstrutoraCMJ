Private Sub Form_Activate()
    DataFim = Date
    Do While Day(DataFim) <= 31
        DataFim = DataFim + 1
       If Month(DataFim) <> Month(Date) Then
            DataFim = DataFim - 1
            Exit Do
        End If
    Loop
    DataIni = DateSerial(Year(Date), Month(Date), ((Day(Date)) - (Day(Date)) + 1))
End Sub

End Sub
