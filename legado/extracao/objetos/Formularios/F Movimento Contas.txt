Private Sub CopiarRegistro_Click()
    Dim Mensagem, Estilo, Resposta
    Mensagem = "Confirma copiar registro ?"
    Estilo = vbYesNo + vbQuestion + vbDefaultButton2
    Resposta = MsgBox(Mensagem, Estilo, "CashBook")
    
    If Resposta = vbNo Then Exit Sub
        
    DoCmd.RunCommand acCmdSelectRecord
    DoCmd.RunCommand acCmdCopy
    DoCmd.RunCommand acCmdRecordsGoToNew
    DoCmd.RunCommand acCmdSelectRecord
    DoCmd.RunCommand acCmdPaste

    Mensagem = "Registro foi copiado e está editado para alterações."
    Estilo = vbOK + vbInformation
    Resposta = MsgBox(Mensagem, Estilo, "CashBook")
    
    DoCmd.GoToControl "DataMovto"
End Sub

Private Sub Confirmado_Click()
    Dim EstaConfirmado As Boolean
   
    EstaConfirmado = Confirmado
    If (EstaConfirmado) Then
        Efetuado = True
        If (IsNull(DataPagto)) Then DataPagto = Date
        ' se lançto é de venda atribui os códigos do parceiro ou empresa
        If (CodigoParceiro = 1) Then
            If (CodAtribuiParceiro <> 0) Then
                CodAnteriorParceiro = CodigoParceiro
                CodigoParceiro = CodAtribuiParceiro
            End If
        End If
        If (EmpVinculada = 1) Then
            If (CodAtribuiEmpresa <> 0) Then
                CodAnteriorEmpresa = EmpVinculada
                EmpVinculada = CodAtribuiEmpresa
            End If
        End If
    Else
        If (CodigoParceiro <> 1) Then
            If (CodAnteriorParceiro <> 0) Then
                CodigoParceiro = CodAnteriorParceiro
            End If
        End If
        If (EmpVinculada <> 1) Then
            If (CodAnteriorEmpresa <> 0) Then
                EmpVinculada = CodAnteriorEmpresa
            End If
        End If
    End If

    ' se tem pagto lote atualiza todos os movtos envolvidos para Efetuado = S ou N
    If (MovPagtoLote <> 0) Then
        DoCmd.OpenQuery "AtualizaEfetuadoLote", acViewNormal, acEdit
        DoCmd.OpenQuery "AtualizaConfirmadoLote", acViewNormal, acEdit
    End If
    If (LinkVenda <> 0) Then
        DoCmd.OpenQuery "AtualizaEfetuadoVenda", acViewNormal, acEdit
        DoCmd.OpenQuery "AtualizaConfirmadoVenda", acViewNormal, acEdit
    End If

End Sub

        End If

            If (CodAnteriorEmpresa <> 0) Then
                EmpVinculada = CodAnteriorEmpresa
            End If
        End If

    If (MovPagtoLote <> 0) Then
        DoCmd.OpenQuery "AtualizaEfetuadoLote", acViewNormal, acEdit

    If (LinkVenda <> 0) Then

Private Sub GerarParcelas_Click()
On Error GoTo Err_GerarParcelas_Click

    Dim stDocName As String
    Dim stLinkCriteria As String
    Dim MovLote
    Dim Mensagem, Estilo, Resposta

    Dim QuantidadeParcelas
    Dim DiaVencimento
    Dim QuantidadeDias
    Dim QtdParcelasGeradas
    Dim DataVencimento
    Dim MovPagtoLote
    Dim ComplementoOriginal
    Dim TTParcelasOriginal
    Dim TTParcelasGeradas
    
    QuantidadeParcelas = QtdParcelas
    DiaVencimento = DiaVencto
    QuantidadeDias = QtdDias
    DataVencimento = DataVencto
    
    If IsNull(DataVencimento) Then
        MsgBox "Data de vencimento e obrigatoria !"
        Exit Sub
    End If
    
    If IsNull(QuantidadeParcelas) Then
        MsgBox "Obrigatorio numero de parcelas a gerar !"
        Exit Sub
    End If
    
    If ((IsNull(QuantidadeDias)) Or (QuantidadeDias <= 0) Or (QuantidadeDias > 31)) Then
        MsgBox "Digite a quantidade de dias para o proximo vencimento entre 1 e 31 !"
        Exit Sub
    End If
    
    If IsNull(DiaVencto) Then DiaVencimento = 0
    
    Mensagem = "Confirma gerar parcelas !"
    Estilo = vbYesNo + vbQuestion
    Resposta = MsgBox(Mensagem, Estilo, "Movimento")
    If Resposta = vbNo Then Exit Sub
    
    ' monta a primeira parcela que e o proprio lancamento => Complemento + 1/xx
    Me!MovPagtoLote.Enabled = True
    Me!AbrirAgrupadaLote.Enabled = True
    Me!MovPagtoLote = Me!NumMovimento
    ComplementoOriginal = Me!Complemento
    ComplementoProximaParcela = Trim(ComplementoOriginal) & " 1/" & QuantidadeParcelas
    Me!Complemento = ComplementoProximaParcela
    
    DoCmd.DoMenuItem acFormBar, acRecordsMenu, acSaveRecord, , acMenuVer70
    
    TTParcelasOriginal = QtdParcelas
    QuantidadeParcelas = QuantidadeParcelas - 1
    TTParcelasGeradas = 1
    
    ProximoVencto = DataVencimento + QuantidadeDias
    Do While QtdParcelasGeradas < QuantidadeParcelas
        TTParcelasGeradas = TTParcelasGeradas + 1
        If DiaVencimento <> 0 Then
            If Day(ProximoVencto) <> Day(DataVencimento) Then
                While Day(ProximoVencto) > Day(DataVencimento)
                    ProximoVencto = ProximoVencto - 1
                Wend
                While Day(ProximoVencto) < Day(DataVencimento)
                    ProximoVencto = ProximoVencto + 1
                Wend
            End If
        End If
        ' monta complemento proxima parcela
        ComplementoProximaParcela = Trim(ComplementoOriginal) & " " & TTParcelasGeradas & "/" & TTParcelasOriginal
        ' ativa consulta inclusao no F Temp Movimento Contas
        DoCmd.OpenQuery "Gerar Temp Parcelas Movto", acViewNormal, acAdd
        ' calcula vencto e incrementa parcelas a gerar
        QtdParcelasGeradas = (QtdParcelasGeradas + 1)
        ProximoVencto = ProximoVencto + QuantidadeDias
    Loop
    
    ' gera arquivo F Temp Movimento Contas no arquivo REAL
    DoCmd.OpenQuery "Gerar Temp Parcelas No Movto", acViewNormal, acAdd
    
    ' Zera arquivo F Temp Movimento Contas (ja existe rotina)
    DoCmd.OpenQuery "F Zerar Temp Movimento Contas", acViewNormal, acEdit
    
    ' carrega formulario dos pagamentos em LOTE, gravar em LOTE nas variaveis
    MovLote = Me![MovPagtoLote]
    If MovLote = 0 Then Exit Sub
    
    stDocName = "AgrupadaLoteViaMovto"
    
    stLinkCriteria = "[MovPagtoLote]=" & MovLote
    DoCmd.OpenForm stDocName, , , stLinkCriteria
    
    Me.Requery
    DoCmd.GoToRecord , , acNewRec
    
    
Exit_GerarParcelas_Click:
    Exit Sub

Err_GerarParcelas_Click:
    MsgBox Err.Description
    Resume Exit_GerarParcelas_Click
    
End Sub
