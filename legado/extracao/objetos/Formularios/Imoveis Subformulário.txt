Private Sub AbrirAgrupadaVendas_Click()
On Error GoTo Err_AbrirAgrupadaVendas_Click

    Dim stDocName As String
    Dim stLinkCriteria As String
    Dim MovVenda 'As Integer

    MovVenda = Me![CodImovel]
    If MovVenda = 0 Then Exit Sub
    
    stDocName = "AgrupadaVendasViaMovto"
    
    stLinkCriteria = "[AgrupadorVenda]=" & MovVenda
    DoCmd.OpenForm stDocName, , , stLinkCriteria

Exit_AbrirAgrupadaVendas_Click:
    Exit Sub

Err_AbrirAgrupadaVendas_Click:
    MsgBox Err.Description
    Resume Exit_AbrirAgrupadaVendas_Click

End Sub

Sub Form_Current()

    Dim ParentDocName As String

    On Error Resume Next
    ParentDocName = Me.Parent.Name

    If Err <> 0 Then
        GoTo Form_Current_Exit
    Else
        On Error GoTo Form_Current_Err
        Me.Parent![CondPagtoImovel Subformulário].Requery
        Me.Parent![ParcelasImovel].Requery
        Me.Parent![ParcelasImovelCred].Requery
    End If
    
    If Me!Vendido = True Then
        Me!AbrirAgrupadaVendas.Enabled = True
    Else
        Me!AbrirAgrupadaVendas.Enabled = False
    End If

Form_Current_Exit:
    Exit Sub

Form_Current_Err:
    MsgBox Error$
    Resume Form_Current_Exit

End Sub
