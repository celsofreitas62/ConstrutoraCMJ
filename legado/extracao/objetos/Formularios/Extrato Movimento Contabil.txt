Private Sub Form_Activate()
    DTFinal = Date
    Do While Day(DTFinal) <= 31
        DTFinal = DTFinal + 1
       If Month(DTFinal) <> Month(Date) Then
            DTFinal = DTFinal - 1
            Exit Do
        End If
    Loop
    DTInicial = DateSerial(Year(Date), Month(Date), ((Day(Date)) - (Day(Date)) + 1))
End Sub
