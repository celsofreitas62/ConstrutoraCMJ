Private Sub SimularReajuste_Click()
On Error GoTo Err_SimularReajuste_Click
    
    Dim TotalRegistros
    Dim RegistroLido
    Dim Mensagem, Estilo, Resposta
    
    If PercentualReajuste = 0 Then Exit Sub
    
    If ((IsNull(FatorReajuste)) Or (FatorReajuste <> "%" And FatorReajuste <> "*")) Then
        Mensagem = "Fator de reajuste deve ser igual a % ou *"
        Estilo = vbInformation + vbOKOnly
        Resposta = MsgBox(Mensagem, Estilo, "Reajuste de Parcelas")
        DoCmd.GoToControl "FatorReajuste"
        Exit Sub
    End If
    
    TotalRegistros = DCount("[ValorDebito]", "Reajuste de Parcelas Parceiro")
    
    DoCmd.GoToRecord , , acFirst
    For RegistroLido = 1 To TotalRegistros
        If FatorReajuste = "%" Then
            ValorReajustado = (ValorOriginalImovel + (ValorOriginalImovel * (PercentualReajuste / 100)))
        End If
        If FatorReajuste = "*" Then
            ValorReajustado = (ValorOriginalImovel * PercentualReajuste)
        End If
        If (RegistroLido < TotalRegistros) Then DoCmd.GoToRecord , , acNext
    Next RegistroLido

    DoCmd.GoToRecord , , acFirst
    
    If TotalAcrescimo <> 0 Then ReajustarParcelas.Enabled = True

Exit_SimularReajuste_Click:
    Exit Sub

Err_SimularReajuste_Click:
    MsgBox Err.Description
    Resume Exit_SimularReajuste_Click
    
End Sub
Private Sub ReajustarParcelas_Click()
On Error GoTo Err_ReajustarParcelas_Click

    Dim ValorSelecionado
    Dim TotalSelecionado
    Dim TotalRegistros
    Dim RegistroLido
    Dim Mensagem, Estilo, Resposta
    
    If (IsNull(PercentualReajuste)) Then Exit Sub
        
    ValorSelecionado = TotalSelecionado
    
    Mensagem = "Confirma reajuste !"
    Estilo = vbYesNo + vbQuestion
    Resposta = MsgBox(Mensagem, Estilo, "Reajustar Parcelas")
    If Resposta = vbNo Then Exit Sub
    
    TotalRegistros = DCount("[ValorDebito]", "Reajuste de Parcelas Parceiro")
    
    If TotalRegistros = 0 Then Exit Sub
    
    DoCmd.GoToRecord , , acFirst
    For RegistroLido = 1 To TotalRegistros
        ValorDebito = ValorReajustado
        If (RegistroLido < TotalRegistros) Then DoCmd.GoToRecord , , acNext
    Next RegistroLido

    DoCmd.GoToRecord , , acFirst
    For RegistroLido = 1 To TotalRegistros
        ValorReajustado = 0
        If (RegistroLido < TotalRegistros) Then DoCmd.GoToRecord , , acNext
    Next RegistroLido
    DoCmd.GoToRecord , , acFirst
    
    ' reajusta a parcela da empresa e financeira
    DoCmd.OpenQuery "Reajuste de Parcelas Empresa", acViewNormal, acEdit
        
    DoCmd.GoToControl "PercentualReajuste"
    ReajustarParcelas.Enabled = False
    PercentualReajuste = 0
    
    Mensagem = "Reajuste efetuado com sucesso !"
    Estilo = vbOKOnly + vbInformation
    Resposta = MsgBox(Mensagem, Estilo, "Reajustar Parcelas")

Exit_ReajustarParcelas_Click:
    Exit Sub

Err_ReajustarParcelas_Click:
    MsgBox Err.Description
    Resume Exit_ReajustarParcelas_Click
    
End Sub
